    let checkedState = JSON.parse(localStorage.getItem('baby_vocab_state_300')) || {};


    function initCategories() {

      const select = document.getElementById('category-filter');

      const categories = [...new Set(wordsData.map(w => w.cat))];

      categories.forEach(cat => {

        const option = document.createElement('option');

        option.value = cat;

        option.textContent = cat;

        select.appendChild(option);

      });

      document.getElementById('total-words').textContent = wordsData.length;

    }


    function toggleWord(id) {

      checkedState[id] = !checkedState[id];

      localStorage.setItem('baby_vocab_state_300', JSON.stringify(checkedState));

      renderWords();

    }


    function resetData() {

      if (confirm("Vuoi davvero azzerare tutte le selezioni?")) {

        checkedState = {};

        localStorage.removeItem('baby_vocab_state_300');

        renderWords();

      }

    }


    function renderWords() {

      const grid = document.getElementById('grid');

      const search = document.getElementById('search').value.toLowerCase();

      const selectedCat = document.getElementById('category-filter').value;

      

      grid.innerHTML = '';

      let count = 0;

      

      wordsData.forEach(item => {

        const isChecked = !!checkedState[item.id];

        if (isChecked) count++;


        const matchesSearch = item.name.toLowerCase().includes(search);

        const matchesCat = selectedCat === 'Tutti' || item.cat === selectedCat;


        if (matchesSearch && matchesCat) {

          const card = document.createElement('div');

          card.onclick = () => toggleWord(item.id);

          card.className = `cursor-pointer p-3 rounded-2xl border transition-all flex flex-col items-center justify-between text-center select-none shadow-sm ${

            isChecked 

              ? 'bg-emerald-50 border-emerald-500 ring-2 ring-emerald-400/30' 

              : 'bg-white border-slate-200 hover:border-indigo-300'

          }`;


          card.innerHTML = `

            <div class="text-3xl mb-1">${item.icon}</div>

            <div class="text-xs font-bold ${isChecked ? 'text-emerald-900' : 'text-slate-700'}">${item.name}</div>

            <div class="mt-1.5 text-[10px] font-semibold px-2 py-0.5 rounded-full ${isChecked ? 'bg-emerald-200 text-emerald-800' : 'bg-slate-100 text-slate-400'}">

              ${isChecked ? 'Pronunciata' : 'Da spuntare'}

            </div>

          `;

          grid.appendChild(card);

        }

      });


      document.getElementById('counter').textContent = count;

      const percentage = (count / wordsData.length) * 100;

      document.getElementById('progress-bar').style.width = `${percentage}%`;

    }


    initCategories();

    renderWords();

  </script>
